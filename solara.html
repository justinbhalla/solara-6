}); this.ws.addEventListener('message', (ev) => { let msg; try { msg = JSON.parse(ev.data); } catch (e) { console.warn('invalid msg', ev.data); return; } // persist lastSeenSeq when server sends seqs if (msg.seq && msg.type === 'ops') { this.lastSeenSeq = Math.max(this.lastSeenSeq, msg.seq); } catch (e) { console.error('Realtime connect error', e); setTimeout(() => this._connect(), this._backoff); this._backoff = Math.min(this._backoff * 1.5, this._maxBackoff); } } stop() { this._shouldReconnect = false;
